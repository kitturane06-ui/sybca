const int ledPin = 6; //LED connected to digital pin6
void setup(){
pin mode (led pin, OUTPUT); 1,OUTPUT); // Set pin6 as output
}
void loop(){
digital write (ledpin, HIGH); // Turn LED on 
delay(500); //on Delay 0.5 seconds) 
digital write (ledpin, low); // Turn LED off 
delay (500); // off delay (0.5 seconds)
}
